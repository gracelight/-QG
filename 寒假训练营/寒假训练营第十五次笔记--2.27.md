void UART_Init() { SCON = 0x50; // UART模式1（8位异步）+ REN=1（允许接收） 
		// PCON &= 0x7F;     // SMOD=0（波特率不翻倍，9600波特率核心！） 
TMOD &= 0x0F; // 清空定时器1模式位（低4位保留，高4位清0）
TMOD |= 0x20; // 定时器1设为模式2（8位自动重装，波特率发生器） 
TL1 = 0xFD; // 定时初值（模式2下，TL1会自动重装TH1的值，需和TH1一致） 
TH1 = 0xFD; // 9600波特率重装值（11.0592MHz + SMOD=0） 
ET1 = 0; // 禁止定时器1中断（波特率发生器无需中断，正确） 
TR1 = 1; // 启动定时器1（波特率发生器必须启动，正确） 
EA = 1; // 开总中断（接收中断需要） 
ES = 1; // 开串口中断（接收数据需要，正确） }
